            const CLEAR_COMMANDS = ['hapus layar', 'bersihkan layar', 'slide baru', 'halaman baru', 'mulai dari awal', 'clear screen', 'new slide', 'reset canvas'];

            // ==========================================
            // 3. PARSER & AUTO-LAYOUT ENGINE
            // ==========================================
            function processTranscript(rawText) {
                const text = rawText.trim();
                if (!text) return;

                const lowerText = text.toLowerCase();

                // Check if user wants to clear canvas for a "new slide"
                if (CLEAR_COMMANDS.some(cmd => lowerText.includes(cmd))) {
                    clearScene();
                    showToast('Slide baru dibersihkan');
                    return;
                }

                // Detect all matching concepts in order of appearance in the sentence
                let detected = [];
                CONCEPT_DICTIONARY.forEach(concept => {
                    let earliestIndex = Infinity;
                    let matchedWord = '';
                    concept.keywords.forEach(kw => {
                        // Word boundary-aware search
                        const regex = new RegExp(`\\b${kw}\\b`, 'i');
                        const match = lowerText.match(regex);
                        if (match && match.index < earliestIndex) {
                            earliestIndex = match.index;
                            matchedWord = match[0];
                        }
                    });

                    if (earliestIndex !== Infinity) {
                        detected.push({
                            index: earliestIndex,
                            type: concept.type,
                            label: capitalizeFirst(matchedWord) || concept.defaultLabel
                        });
                    }
                });

                // Sort detected concepts by order spoken
                detected.sort((a, b) => a.index - b.index);

                // If no specific icon keyword found, create a minimalist "Note / Quote Box" node if sentence has >= 2 words
                if (detected.length === 0) {
                    const words = text.split(/\s+/);
                    if (words.length >= 2) {
                        const shortLabel = words.slice(0, 4).join(' ') + (words.length > 4 ? '...' : '');
                        detected.push({
                            index: 0,
                            type: 'note',
                            label: capitalizeFirst(shortLabel)
                        });
                    } else {
                        return;
                    }
                }

                // Add nodes sequentially & link them with hand-drawn arrows
                detected.forEach(item => {
                    // Avoid duplicate consecutive identical icon + label
                    const lastNode = nodes[nodes.length - 1];
                    if (lastNode && lastNode.type === item.type && lastNode.label.toLowerCase() === item.label.toLowerCase()) {
                        return;
                    }

                    const pos = calculateNextNodePosition();
                    const newNode = {
                        id: 'node_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5),
                        type: item.type,
                        label: item.label,
                        x: pos.x,
                        y: pos.y,
                        progress: 0, // 0 to 1 for stroke animation
                        seed: Math.floor(Math.random() * 10000)
                    };

                    nodes.push(newNode);

                    // Connect from previous node if it's on the same row or nearby
                    if (lastNode) {
                        connectors.push({
                            fromId: lastNode.id,
                            toId: newNode.id,
                            progress: 0,
                            seed: Math.floor(Math.random() * 10000)
                        });
                    }

                    // Auto-pan camera smoothly if new node goes off-screen
                    autoFocusCamera(newNode);
                });

                updateEmptyState();
            }

            function capitalizeFirst(str) {
                if (!str) return '';
                return str.charAt(0).toUpperCase() + str.slice(1);
            }

            /**
             * Calculates clean non-overlapping grid-flow coordinates
             * Arranges items left-to-right in rows of 3 or 4 depending on screen width.
             */
            function calculateNextNodePosition() {
                const rect = container.getBoundingClientRect();
                const isMobile = rect.width < 640;
                const cols = isMobile ? 2 : 4;
                const spacingX = isMobile ? 165 : 220;
                const spacingY = isMobile ? 175 : 195;

                const index = nodes.length;
                const col = index % cols;
                const row = Math.floor(index / cols);

                // Center the grid horizontally around canvas center
                const totalRowWidth = (cols - 1) * spacingX;
                const startX = (rect.width / 2) - (totalRowWidth / 2);
                const startY = isMobile ? 150 : 180;

                // Slight organic offset so it feels natural
                const jitterX = ((index * 17) % 12) - 6;
                const jitterY = ((index * 23) % 12) - 6;

                return {
                    x: startX + col * spacingX + jitterX,
                    y: startY + row * spacingY + jitterY
                };
            }

            function autoFocusCamera(node) {
                const rect = container.getBoundingClientRect();
                const screenY = node.y * camera.zoom + camera.y;
                if (screenY > rect.height - 180) {
                    camera.y -= 180;
                }
            }

            function updateEmptyState() {
                if (nodes.length > 0) {
                    emptyHint.classList.add('opacity-0');
                } else {
                    emptyHint.classList.remove('opacity-0');
                }
            }

            // ==========================================
            // 4. MINIMALIST LINE-ART VECTOR RENDERER
            // ==========================================
            function drawMinimalistIcon(node) {
                const { x, y, type, progress, seed, label } = node;
                const p = Math.min(1, progress);
                if (p <= 0) return;

                // Base rough options: clean black ink, minimalist strokes
                const opts = {
                    stroke: '#111111',
                    strokeWidth: 2,
                    roughness: 1.15,
                    bowing: 1.1,
                    seed: seed
                };

                const thinOpts = {
                    ...opts,
                    strokeWidth: 1.5,
                    roughness: 0.9
                };

                ctx.save();
                // Subtle scale-in + ink fade effect
                const scale = 0.85 + 0.15 * easeOutCubic(p);
                ctx.translate(x, y);
                ctx.scale(scale, scale);
                ctx.globalAlpha = Math.min(1, p * 1.4);

                switch (type) {
                    case 'idea':
                        // Minimalist Lightbulb
                        rc.circle(0, -14, 44, opts);
                        rc.line(-10, 6, 10, 6, opts);
                        rc.line(-8, 12, 8, 12, opts);
                        rc.line(-5, 18, 5, 18, opts);
                        // Filament
                        rc.path('M -6 -8 Q 0 -18 6 -8', thinOpts);
                        // Rays (appear in second half of animation)
                        if (p > 0.5) {
                            rc.line(0, -44, 0, -52, thinOpts);
                            rc.line(-28, -32, -35, -38, thinOpts);
                            rc.line(28, -32, 35, -38, thinOpts);
                            rc.line(-34, -12, -43, -12, thinOpts);
                            rc.line(34, -12, 43, -12, thinOpts);
                        }
                        break;

                    case 'team':
                        // Minimalist Group of People (Center + Left/Right)
                        rc.circle(0, -22, 20, opts);
                        rc.path('M -20 14 C -20 -4, 20 -4, 20 14 Z', opts);
                        if (p > 0.45) {
                            rc.circle(-24, -16, 14, thinOpts);
                            rc.path('M -38 14 C -38 2, -16 2, -16 14', thinOpts);
                            rc.circle(24, -16, 14, thinOpts);
                            rc.path('M 16 14 C 16 2, 38 2, 38 14', thinOpts);
                        }
                        break;

                    case 'growth':
                        // Minimalist Axes + Rising Trend Line
                        rc.line(-32, -32, -32, 20, opts);
                        rc.line(-32, 20, 34, 20, opts);
                        if (p > 0.35) {
                            rc.path('M -26 12 L -8 -2 L 6 4 L 26 -24', { ...opts, strokeWidth: 2.4 });
                            // Arrow tip
                            rc.line(26, -24, 15, -23, opts);
                            rc.line(26, -24, 25, -13, opts);
                        }
                        break;

                    case 'money':
                        // Minimalist Coin / Dollar / Value Circle
                        rc.circle(0, -6, 52, opts);
                        rc.circle(0, -6, 42, thinOpts);
                        if (p > 0.4) {
                            // Stylized S / $ vertical bar
                            rc.line(0, -24, 0, 12, opts);
                            rc.path('M 9 -16 C -10 -20, -10 -6, 0 -6 C 10 -6, 10 8, -9 4', { ...opts, strokeWidth: 2.2 });
                        }
                        break;

                    case 'server':

            /**
             * Draws a hand-drawn arrow connecting two nodes
             */
            function drawConnector(conn) {
                const fromNode = nodes.find(n => n.id === conn.fromId);
                const toNode = nodes.find(n => n.id === conn.toId);
                if (!fromNode || !toNode) return;

                const p = Math.min(1, conn.progress);
                if (p <= 0) return;

                const dx = toNode.x - fromNode.x;
                const dy = toNode.y - fromNode.y;
                const dist = Math.hypot(dx, dy);
                if (dist < 95) return;

                const angle = Math.atan2(dy, dx);
                const padStart = 52;
                const padEnd = 52;

                const x1 = fromNode.x + Math.cos(angle) * padStart;
                const y1 = fromNode.y + Math.sin(angle) * padStart;
                const fullX2 = toNode.x - Math.cos(angle) * padEnd;
                const fullY2 = toNode.y - Math.sin(angle) * padEnd;

                // Interpolate endpoint based on animation progress
                const curX2 = x1 + (fullX2 - x1) * easeOutCubic(p);
                const curY2 = y1 + (fullY2 - y1) * easeOutCubic(p);

                const opts = {
                    stroke: '#27272a',
                    strokeWidth: 1.6,
                    roughness: 1.2,
                    seed: conn.seed
                };

                // Slight curved arc for natural whiteboard flow
                const midX = (x1 + curX2) / 2 - Math.sin(angle) * 10;
                const midY = (y1 + curY2) / 2 + Math.cos(angle) * 10;

                rc.path(`M ${x1} ${y1} Q ${midX} ${midY} ${curX2} ${curY2}`, opts);

                // Arrowhead once line reaches >= 85%
                if (p >= 0.85) {
                    const headLen = 10;
                    const endAngle = Math.atan2(curY2 - midY, curX2 - midX);
                    const ax1 = curX2 - headLen * Math.cos(endAngle - Math.PI / 6);
                    const ay1 = curY2 - headLen * Math.sin(endAngle - Math.PI / 6);
                    const ax2 = curX2 - headLen * Math.cos(endAngle + Math.PI / 6);
                    const ay2 = curY2 - headLen * Math.sin(endAngle + Math.PI / 6);

                    rc.line(curX2, curY2, ax1, ay1, opts);
                    rc.line(curX2, curY2, ax2, ay2, opts);
                }
            }

            function easeOutCubic(t) {
                return 1 - Math.pow(1 - t, 3);
            }

            // ==========================================
            // 5. MAIN ANIMATION RENDER LOOP
            // ==========================================
            let lastTime = performance.now();

            function renderLoop(now) {
                const dt = (now - lastTime) / 1000;
                lastTime = now;

                const rect = container.getBoundingClientRect();
                ctx.clearRect(0, 0, rect.width, rect.height);

                ctx.save();
                ctx.translate(camera.x, camera.y);
                ctx.scale(camera.zoom, camera.zoom);

                // Animate & render connectors
                connectors.forEach(conn => {
                    if (conn.progress < 1) {
                        conn.progress = Math.min(1, conn.progress + dt * 2.4);
                    }
                    drawConnector(conn);
                });

                // Animate & render line-art nodes
                nodes.forEach(node => {
                    if (node.progress < 1) {
                        node.progress = Math.min(1, node.progress + dt * 2.2);
                    }
                    drawMinimalistIcon(node);
                });

                ctx.restore();

                requestAnimationFrame(renderLoop);
            }
            requestAnimationFrame(renderLoop);

            // ==========================================
            // 6. WEB SPEECH API (LIVE MICROPHONE)
            // ==========================================
            function initSpeechRecognition() {
                const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
                if (!SpeechRecognition) {
                    liveInterimText.textContent = 'Browser tidak mendukung Web Speech API. Gunakan input teks di bawah.';
                    return null;
                }

                const recog = new SpeechRecognition();
                recog.continuous = true;
                recog.interimResults = true;
                recog.lang = currentLang;

                recog.onstart = () => {
                    isListening = true;
                    micBtn.classList.add('mic-active', 'bg-zinc-900', 'text-white');
                    micBtn.classList.remove('bg-zinc-100', 'text-zinc-900');
                    liveInterimText.textContent = currentLang === 'id-ID' ? 'Mendengarkan... silakan bicara' : 'Listening... speak now';
                };

                recog.onresult = (event) => {
                    let interimTranscript = '';
                    let finalTranscript = '';

                    for (let i = event.resultIndex; i < event.results.length; ++i) {
                        const transcript = event.results[i][0].transcript;
                        if (event.results[i].isFinal) {
                            finalTranscript += transcript;
                        } else {
                            interimTranscript += transcript;
                        }
                    }

                    if (interimTranscript) {
                        liveInterimText.textContent = `"${interimTranscript}..."`;
                    }

                    if (finalTranscript) {
                        liveInterimText.textContent = `✓ "${finalTranscript.trim()}"`;
                        processTranscript(finalTranscript);
                    }
                };

                recog.onerror = (event) => {
                    if (event.error === 'not-allowed') {
                        showToast('Izin mikrofon ditolak. Gunakan kotak teks untuk mencoba.');
                        stopListening();
                    } else if (event.error !== 'no-speech') {
                        liveInterimText.textContent = `Mic status: ${event.error}`;
                    }
                };

                recog.onend = () => {
                    // Auto-restart if still in active presentation mode
                    if (isListening) {
                        try {
                            recog.start();
                        } catch (e) {
                            stopListening();
                        }
                    }
                };

                return recog;
            }

            function startListening() {
                if (!recognition) {
                    recognition = initSpeechRecognition();
                }
                if (!recognition) {
                    showToast('Browser tidak mendukung input suara langsung.');
                    return;
                }
                recognition.lang = currentLang;
                try {
                    recognition.start();
                } catch (e) {
                    // Already started
                }
            }

            function stopListening() {
                isListening = false;
                micBtn.classList.remove('mic-active', 'bg-zinc-900', 'text-white');
                micBtn.classList.add('bg-zinc-100', 'text-zinc-900');
                liveInterimText.textContent = 'Mikrofon dijeda.';
                if (recognition) {
                    try {
                        recognition.stop();
                    } catch (e) {}
                }
            }

            micBtn.addEventListener('click', () => {
                if (isListening) {
                    stopListening();
                } else {
                    startListening();
                }
            });
